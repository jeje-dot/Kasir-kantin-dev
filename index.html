// State Global Aplikasi let activeUser = null; let productsData = []; let cart = []; // Seed Default Data untuk testing (Jika DB Kosong) async function seedDatabase() { const usersRef = ref(db, 'users'); const snap = await get(usersRef); if (!snap.exists()) { await set(ref(db, 'users/123456'), { name: "Jefri", role: "owner", tenantId: "ALL" }); await set(ref(db, 'users/111111'), { name: "Kasir Ayam Geprek", role: "tenant", tenantId: "T01" }); await set(ref(db, 'products/P01'), { name: "Ayam Geprek", price: 20000, stock: 50, tenantId: "T01", img: "fa-drumstick-bite" }); await set(ref(db, 'products/P02'), { name: "Es Teh Manis", price: 5000, stock: 100, tenantId: "T01", img: "fa-glass-whiskey" }); await set(ref(db, 'products/P03'), { name: "Mie Ayam", price: 18000, stock: 30, tenantId: "T02", img: "fa-bowl-food" }); } } seedDatabase(); // ========================================== // LOGIKA AUTENTIKASI (OWNER VS TENANT) // ========================================== window.authLogin = async function() { const pin = document.getElementById('input-pin').value; if(!pin) return; try { const userRef = ref(db, 'users/' + pin); const snapshot = await get(userRef); if (snapshot.exists()) { activeUser = snapshot.val(); document.getElementById('login-screen').style.display = 'none'; document.getElementById('app-container').style.display = 'flex'; document.getElementById('lbl-user-name').innerText = activeUser.name; document.getElementById('lbl-tenant-name').innerText = activeUser.role === 'owner' ? "Pusat (Semua Akses)" : "Kios " + activeUser.tenantId; // Kontrol Visibilitas Menu if(activeUser.role === 'owner') { document.getElementById('owner-menus').style.display = 'block'; } else { document.getElementById('owner-menus').style.display = 'none'; // Tenant tidak butuh bottom panel yg lengkap, atau disesuaikan } loadProducts(); } else { Swal.fire('Gagal', 'PIN tidak ditemukan di database server!', 'error'); } } catch (error) { console.error("Firebase Error:", error); Swal.fire('Error', 'Gagal koneksi ke database. Pastikan Config benar.', 'error'); } } // ========================================== // LOAD PRODUK & RENDER // ========================================== function loadProducts() { const prodRef = ref(db, 'products'); onValue(prodRef, (snapshot) => { productsData = []; snapshot.forEach((child) => { let p = child.val(); p.id = child.key; // Filter: Jika tenant, hanya load barang tokonya. Jika owner, bebas. if (activeUser.role === 'owner' || p.tenantId === activeUser.tenantId) { productsData.push(p); } }); renderProducts(productsData); }); } function renderProducts(dataArray) { const container = document.getElementById('product-grid-container'); container.innerHTML = ''; dataArray.forEach(p => { container.innerHTML += `
Stok: ${p.stock}
${p.name}
Rp ${p.price.toLocaleString('id-ID')}
`; }); } window.filterProducts = function() { const keyword = document.getElementById('search-input').value.toLowerCase(); const filtered = productsData.filter(p => p.name.toLowerCase().includes(keyword)); renderProducts(filtered); } // ========================================== // KERANJANG BELANJA (CART) // ========================================== window.addToCart = function(id) { const p = productsData.find(x => x.id === id); if(!p) return; const existing = cart.find(x => x.id === id); if(existing) { if(existing.qty < p.stock) existing.qty++; else Swal.fire('Limit', 'Stok habis', 'warning'); } else { cart.push({...p, qty: 1}); } renderCart(); } window.changeQty = function(id, delta) { const item = cart.find(x => x.id === id); if(item) { item.qty += delta; if(item.qty <= 0) cart = cart.filter(x => x.id !== id); renderCart(); } } function renderCart() { const container = document.getElementById('cart-items-container'); let total = 0; let count = 0; container.innerHTML = ''; if(cart.length === 0) { container.innerHTML = `
Keranjang kosong
`; } else { cart.forEach(item => { let sub = item.price * item.qty; total += sub; count += item.qty; container.innerHTML += `

${item.name}

Rp ${item.price.toLocaleString('id-ID')}
${item.qty}
`; }); } document.getElementById('cart-count').innerText = `${count} Item`; document.getElementById('cart-subtotal').innerText = `Rp ${total.toLocaleString('id-ID')}`; document.getElementById('cart-total').innerText = `Rp ${total.toLocaleString('id-ID')}`; } // ========================================== // CHECKOUT & LOKAL PRINTING (BUKAN BLUETOOTH MULTI) // ========================================== window.prosesCheckout = async function() { if(cart.length === 0) return; let total = cart.reduce((sum, item) => sum + (item.price * item.qty), 0); // Simpan transaksi ke Firebase const trxRef = ref(db, 'transactions'); const newTrxRef = push(trxRef); const trxData = { idNota: "INV-" + Date.now().toString().slice(-6), waktu: new Date().toISOString(), kasir: activeUser.name, tenantId: activeUser.tenantId, total: total, items: cart.map(i => ({id: i.id, name: i.name, qty: i.qty, price: i.price})) }; try { await set(newTrxRef, trxData); // Update Stok cart.forEach(async item => { const stRef = ref(db, `products/${item.id}/stock`); await set(stRef, item.stock - item.qty); }); // Cetak Struk (Lokal ke Device masing-masing tenant) cetakLokal(trxData); cart = []; renderCart(); Swal.fire('Berhasil', 'Transaksi diproses & Disimpan ke Cloud!', 'success'); } catch (err) { Swal.fire('Error', 'Gagal menyimpan transaksi', 'error'); } } function cetakLokal(trx) { // Ini akan memicu dialog print standar dari browser / tablet POS // Menghindari konflik bluetooth multi-tenant. Masing-masing tablet handle routernya sendiri let area = document.getElementById('print-area'); let itemsHtml = trx.items.map(i => `
${i.name} x${i.qty} Rp ${(i.price * i.qty).toLocaleString('id-ID')}
`).join(''); area.innerHTML = `

PUJASERA POS

Nota: ${trx.idNota}

Tgl: ${new Date().toLocaleString('id-ID')}

Kios: ${trx.tenantId}

${itemsHtml}
TOTAL Rp ${trx.total.toLocaleString('id-ID')}
`; // Eksekusi print window.print(); }
